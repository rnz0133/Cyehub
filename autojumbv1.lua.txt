local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")
local player = Players.LocalPlayer
local character = player.Character or player.CharacterAdded:Wait()
local hrp = character:WaitForChild("HumanoidRootPart")

-- Biến trạng thái
local autoJumpEnabled = false
local jumpDelay = 0.08 -- Tốc độ mặc định (giây)

---------------------------------------------------------
-- 1. GIẢ LẬP CLICK NÚT TRONG GAME (UP / TURN)
---------------------------------------------------------
local function triggerButton(btnName)
    local playerGui = player:WaitForChild("PlayerGui")
    for _, gui in pairs(playerGui:GetChildren()) do
        if gui:IsA("ScreenGui") and gui.Name ~= "AutoJumpMenuGui" then
            local btn = gui:FindFirstChild(btnName, true)
            if btn and btn:IsA("GuiButton") then
                for _, conn in pairs(getconnections(btn.MouseButton1Click or btn.Activated)) do
                    conn:Fire()
                end
                return true
            end
        end
    end
    return false
end

---------------------------------------------------------
-- 2. THUẬT TOÁN CHECK BẬC CẦU THANG (RAYCAST)
---------------------------------------------------------
local function checkStepAhead()
    if not hrp or not hrp.Parent then return false end
    local raycastParams = RaycastParams.new()
    raycastParams.FilterAncestorsInstances = {character}
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude

    local forwardVector = hrp.CFrame.LookVector
    local rayOrigin = hrp.Position + Vector3.new(0, 1, 0)
    local rayDirection = (forwardVector * 3) + Vector3.new(0, 2, 0)

    local result = workspace:Raycast(rayOrigin, rayDirection, raycastParams)
    return result ~= nil
end

---------------------------------------------------------
-- 4. XỬ LÝ SỰ KIỆN GIAO DIỆN
---------------------------------------------------------
-- Mở / Đóng Menu
openBtn.MouseButton1Click:Connect(function()
    mainFrame.Visible = not mainFrame.Visible
end)

closeBtn.MouseButton1Click:Connect(function()
    mainFrame.Visible = false
end)

-- Sự kiện bấm nút Bật/Tắt
toggleBtn.MouseButton1Click:Connect(function()
    autoJumpEnabled = not autoJumpEnabled
    if autoJumpEnabled then
        toggleBtn.Text = "ON"
        toggleBtn.BackgroundColor3 = Color3.fromRGB(45, 180, 80) -- Xanh khi Bật
    else
        toggleBtn.Text = "OFF"
        toggleBtn.BackgroundColor3 = Color3.fromRGB(220, 50, 50)
    end
end)

-- Xử lý đổi tốc độ nhảy
speedInput.FocusLost:Connect(function()
    local val = tonumber(speedInput.Text)
    if val and val >= 0.01 then
        jumpDelay = val
    else
        speedInput.Text = tostring(jumpDelay) -- Reset lại nếu nhập lỗi
    end
end)

---------------------------------------------------------
-- 5. VÒNG LẶP THỰC THI AUTO JUMP
---------------------------------------------------------
task.spawn(function()
    while true do
        if autoJumpEnabled then
            local hasNextStepForward = checkStepAhead()
            if hasNextStepForward then
                triggerButton("up") or triggerButton("Up")
            else
                triggerButton("turn") or triggerButton("Turn")
            end
        end
        task.wait(jumpDelay)
    end
end)
