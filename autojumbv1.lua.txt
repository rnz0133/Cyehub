local Players = game:GetService("Players")
local player = Players.LocalPlayer
local character = player.Character or player.CharacterAdded:Wait()
local hrp = character:WaitForChild("HumanoidRootPart")

-- Biến cấu hình
local autoJumpEnabled = false
local jumpDelay = 0.08

-- Cập nhật Character khi respawn
player.CharacterAdded:Connect(function(newChar)
    character = newChar
    hrp = character:WaitForChild("HumanoidRootPart")
end)

-- Tối ưu vị trí chứa GUI tương thích Executor
local parentGui
if gethui then
    parentGui = gethui()
elseif syn and syn.protect_gui then
    local gui = Instance.new("ScreenGui")
    syn.protect_gui(gui)
    gui.Parent = game:GetService("CoreGui")
    parentGui = gui
elseif game:GetService("CoreGui"):FindFirstChild("RobloxGui") then
    parentGui = game:GetService("CoreGui")
else
    parentGui = player:WaitForChild("PlayerGui")
end

if parentGui:FindFirstChild("AutoJumpMenuGui") then
    parentGui:FindFirstChild("AutoJumpMenuGui"):Destroy()
end

---------------------------------------------------------
-- 1. GIẢ LẬP BẤM NÚT UI (SỬ DỤNG GETCONNECTIONS)
---------------------------------------------------------
local function pressGameButton(btnName)
    local playerGui = player:FindFirstChild("PlayerGui")
    if not playerGui then return false end

    for _, gui in pairs(playerGui:GetChildren()) do
        if gui:IsA("ScreenGui") and gui.Name ~= "AutoJumpMenuGui" then
            local btn = gui:FindFirstChild(btnName, true)
            if btn and btn:IsA("GuiButton") then
                -- Dùng getconnections để trigger event của Executor
                if getconnections then
                    for _, conn in pairs(getconnections(btn.MouseButton1Click)) do
                        conn:Fire()
                    end
                    for _, conn in pairs(getconnections(btn.Activated)) do
                        conn:Fire()
                    end
                else
                    btn:Activate()
                end
                return true
            end
        end
    end
    return false
end

---------------------------------------------------------
-- 2. RAYCAST CHECK BẬC CẦU THANG (LOGIC SỬA BỞI CLAUDE)
---------------------------------------------------------
local function checkStepAhead()
    if not hrp or not hrp.Parent then return false end

    local raycastParams = RaycastParams.new()
    raycastParams.FilterDescendantsInstances = {character}
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude

    local forwardVector = hrp.CFrame.LookVector
    local rayOrigin = hrp.Position

    -- Tia thấp: Kiểm tra bậc trước mặt (tầm ngang đầu gối)
    local lowRay = workspace:Raycast(
        rayOrigin + Vector3.new(0, -0.5, 0),
        forwardVector * 3.2,
        raycastParams
    )

    if not lowRay then
        return false 
    end

    -- Tia cao: Kiểm tra xem có bị chắn đầu không
    local highRay = workspace:Raycast(
        rayOrigin + Vector3.new(0, 2, 0),
        forwardVector * 3.2,
        raycastParams
    )

    return highRay == nil
end

---------------------------------------------------------
-- 4. SỰ KIỆN GIAO DIỆN
---------------------------------------------------------
openBtn.MouseButton1Click:Connect(function()
    mainFrame.Visible = not mainFrame.Visible
end)

closeBtn.MouseButton1Click:Connect(function()
    mainFrame.Visible = false
end)

toggleBtn.MouseButton1Click:Connect(function()
    autoJumpEnabled = not autoJumpEnabled
    toggleBtn.Text = autoJumpEnabled and "ON" or "OFF"
    toggleBtn.BackgroundColor3 = autoJumpEnabled and Color3.fromRGB(45, 180, 80) or Color3.fromRGB(220, 50, 50)
end)

speedInput.FocusLost:Connect(function()
    local val = tonumber(speedInput.Text)
    if val and val >= 0.01 then
        jumpDelay = val
    else
        speedInput.Text = tostring(jumpDelay)
    end
end)

---------------------------------------------------------
-- 5. VÒNG LẶP XỬ LÝ DI CHUYỂN
---------------------------------------------------------
task.spawn(function()
    while true do
        if autoJumpEnabled then
            local hasNextStepForward = checkStepAhead()
            if hasNextStepForward then
                pressGameButton("up") or pressGameButton("Up")
            else
                pressGameButton("turn") or pressGameButton("Turn")
            end
        end
        task.wait(jumpDelay)
    end
end)
